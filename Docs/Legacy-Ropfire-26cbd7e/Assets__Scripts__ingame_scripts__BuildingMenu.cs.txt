using UnityEngine;
using UnityEngine.UI;

public class BuildingMenu : MonoBehaviour
{
    [Header("Панель меню строительства")]
    public GameObject buildPanel;

    [Header("Префаб ячейки постройки")]
    public GameObject slotPrefab;

    [Header("Куда спавнить ячейки (объект Content внутри Scroll View)")]
    public Transform contentArea;

    [Header("Список доступных зданий")]
    public BuildingData[] buildings;

    private void Start()
    {
        // Создаем кнопки для всех построек из списка
        PopulateMenu();
    }

    private void PopulateMenu()
    {
        if (contentArea == null) return;

        // Если в сцене кнопка-шаблон лежит прямо внутри Content:
        // прячем сам шаблон, чтобы он не удалялся скриптом Destroy()!
        if (slotPrefab != null && slotPrefab.transform.IsChildOf(contentArea))
        {
            slotPrefab.SetActive(false);
        }

        // Очищаем предыдущие созданные копии (не трогая сам шаблон)
        for (int i = contentArea.childCount - 1; i >= 0; i--)
        {
            Transform child = contentArea.GetChild(i);
            if (slotPrefab != null && child.gameObject == slotPrefab)
                continue;
            Destroy(child.gameObject);
        }

        if (slotPrefab == null) return;

        // Создаем кнопку под каждое здание из массива
        for (int i = 0; i < buildings.Length; i++)
        {
            BuildingData data = buildings[i];
            GameObject newSlot = Instantiate(slotPrefab, contentArea);
            newSlot.SetActive(true); // делаем видимой копию
            newSlot.name = "Slot_" + data.buildingName;

            // Находим текст с названием здания (если есть)
            Text titleText = newSlot.GetComponentInChildren<Text>();
            if (titleText != null)
            {
                titleText.text = data.buildingName;
            }

            // Навешиваем клик на кнопку
            Button btn = newSlot.GetComponent<Button>();
            if (btn != null)
            {
                BuildingData chosenBuilding = data;
                btn.onClick.AddListener(() => OnSelectBuilding(chosenBuilding));
            }
        }
    }

    // Вызывается при клике на конкретное здание
    public void OnSelectBuilding(BuildingData building)
    {
        Debug.Log("Выбрано здание для постройки: " + building.buildingName + 
                  " (Цена: " + building.woodCost + " дерева, " + building.beerCost + " пива)");
        
        // Тут передаем команду в систему строительства / курсор
    }

    // Открыть/закрыть меню по клику на вкладку "Строить"
    public void ToggleBuildMenu()
    {
        if (buildPanel != null)
        {
            buildPanel.SetActive(!buildPanel.activeSelf);
        }
    }
}

// Простой класс с данными о здании
[System.Serializable]
public class BuildingData
{
    public string buildingName = "Башня";
    public Sprite icon;
    public int woodCost = 10;
    public int beerCost = 5;
    public GameObject buildingPrefab;
}
