using UnityEngine;
using UnityEngine.UI;

public class ButtonBuildSetting : MonoBehaviour
{
    [Header("Данные постройки (из BuildManager)")]
    public BuildSetting setting;

    [Header("Или префаб напрямую (для ручной настройки)")]
    public GameObject buildingPrefab;

    public Image icon;

    void Start()
    {
        if (setting != null)
        {
            if (setting.icon != null && icon != null)
                icon.sprite = setting.icon;

            if (setting.buildPref != null)
                buildingPrefab = setting.buildPref;
        }
    }

    // Вызывается по клику на кнопку в UI
    public void SpawnBuilding()
    {
        GameObject prefabToSpawn = buildingPrefab;
        if (prefabToSpawn == null && setting != null)
        {
            prefabToSpawn = setting.buildPref;
        }

        if (prefabToSpawn != null)
        {
            // Создаем здание — оно сразу привяжется к мышке
            Instantiate(prefabToSpawn);
        }
    }
}
